# camoufox + playwright pins live in shared/images/base-camoufox.Dockerfile
# (camoufox[geoip]>=0.4.11,<0.5 + playwright==1.49.0). Bump them there;
# this collector picks them up via FROM wealthdb/base-camoufox. The pin
# matters: camoufox-py 0.4.11 downloads Firefox 135.0.1-beta.24 whose
# juggler patches target Playwright 1.49.x — a newer Playwright crashes
# the camoufox-launched Firefox.

# pypdfium2 ships Python bindings to Google's PDFium (the
# renderer Chrome uses). Pure-C++ under the hood, an order of
# magnitude faster than the pdfplumber / pdfminer.six pure-Python
# stack we used originally — pdf_parsers.py does text-only
# extraction across hundreds of monthly statement PDFs, and that
# was CPU-bound on Python-loop overhead. Apache/BSD-licensed,
# wheel-distributed.
pypdfium2>=4.30,<5

# pytest is bundled in the image so tests can be run inside the
# container with `./schwab-web sh -c "cd /app && pytest"`,
# no host-side Python toolchain required.
pytest>=8,<9
