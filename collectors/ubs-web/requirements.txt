# The base image (mcr.microsoft.com/playwright/python:v1.59.0-noble)
# ships only the *browser binaries* under /ms-playwright (with the
# PLAYWRIGHT_BROWSERS_PATH env var pointing at them). The Python
# `playwright` package itself must be installed here, and the version
# MUST match the base-image tag so the package finds compatible
# browser binaries. When bumping the Dockerfile FROM line, bump this
# version in lockstep.
playwright==1.59.0

# Pillow renders the Access-App QR PNG to the terminal as Unicode
# half-blocks. The Playwright base image happens to ship a recent
# Pillow already; pinning here so we are not at the mercy of base-
# image changes.
Pillow>=10.0,<12

# pdfplumber sits on top of pdfminer.six and exposes a clean table-
# extraction API. Used by load.py to reconstruct historical position
# snapshots from UBS's semi-annual "Statement of assets" PDFs and
# month-end "Account Statement" PDFs in the documents archive.
#
# schwab-web migrated its archive to pypdfium2 (order-of-magnitude
# faster) but ubs-web stays on pdfplumber: PDFium's text rects are
# either cell-level (a single row splits into many "lines") or
# baseline-shared across columns (rows that pdfplumber kept apart
# get glued together), and round-tripping the existing line-anchored
# regex layer would need substantial parser surgery. See
# pdf_parsers.py's module docstring for the empirical findings.
pdfplumber>=0.11,<1

# pytest is bundled in the image so tests can be run inside the
# container with `./ubs-web sh -c "cd /app && pytest"`, no
# host-side Python toolchain required.
pytest>=8,<9
