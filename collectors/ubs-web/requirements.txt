# The base image (mcr.microsoft.com/playwright/python:v1.65.2-noble)
# ships only the *browser binaries* under /ms-playwright (with the
# PLAYWRIGHT_BROWSERS_PATH env var pointing at them). The Python
# `playwright` package itself must be installed here, and the version
# MUST match the base-image tag so the package finds compatible
# browser binaries. `make update` moves the tag and this pin together.
playwright==1.65.2

# Pillow renders the Access-App QR PNG to the terminal as Unicode
# half-blocks. The Playwright base image happens to ship a recent
# Pillow already; the pin here keeps a base-image change from moving
# it.
Pillow>=10.0,<12

# pdfplumber (on pdfminer.six) extracts the documents archive's text
# and word positions; see pdf_parsers.py.
#
# pdfplumber, not pypdfium2: see pdf_parsers.py's module docstring.
pdfplumber>=0.11,<1

# pytest is bundled in the image: `make test-ubs-web` runs the suite there, on
# a copy of the checkout, so no host-side Python toolchain is needed.
pytest>=8,<9
