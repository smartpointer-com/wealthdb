# The base image (mcr.microsoft.com/playwright/python:v1.64.1-noble)
# ships only the *browser binaries* under /ms-playwright (with the
# PLAYWRIGHT_BROWSERS_PATH env var pointing at them). The Python
# `playwright` package itself must be installed here, and the version
# MUST match the base-image tag so the package finds compatible
# browser binaries. `make update` moves the tag and this pin together.
playwright==1.64.1

# Swissquote's positions export is a CDFV2 Microsoft Excel file (the
# legacy binary .xls format, not modern .xlsx). xlrd reads .xls;
# openpyxl does not. If Swissquote ever switches to .xlsx, drop xlrd
# and switch the loader to openpyxl.
xlrd==1.2.0

# pypdf is used by load.py to extract position snapshots from
# Portfolio performance PDFs.
pypdf>=6.19,<7

# fontTools lets pypdf decode the CFF Type1 fonts those PDFs embed.
# Without it, pypdf logs a warning for every such font.
fonttools>=4.66,<5

# pytest is bundled in the image: `make test-swissquote` runs the suite there, on
# a copy of the checkout, so no host-side Python toolchain is needed.
pytest>=8,<9

# xlwt writes the legacy binary .xls format xlrd reads. Test-only:
# the XLS parser tests synthesise Positions / List of Assets
# workbooks with it instead of shipping binary fixtures.
xlwt==1.3.0
